📝 Copy this script:
function doPost(e) {
  var data = JSON.parse(e.postData.contents);
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  sheet.appendRow([data.date, data.shopName, data.phone, data.booker, data.timestamp]);
  return ContentService.createTextOutput("OK");
}
function doGet() { return ContentService.createTextOutput("Service is running"); }
